struct FastDiv { unsigned int mul; unsigned int shr; int div; };
inline FastDiv make_fastdiv(int denom) {
    FastDiv f; f.mul = 0u; f.shr = 0u; f.div = denom;
    if (denom <= 1) return f;
    const unsigned int d = (unsigned int)denom;
    int a = 0; while (a < 31 && (1u << a) < d) ++a;        // ceil(log2(d)) == CUTLASS find_log2
    const unsigned int p = 31u + (unsigned int)a;
    f.mul = (unsigned int)((((unsigned long long)1 << p) + (unsigned long long)d - 1ull) / (unsigned long long)d);
    f.shr = p - 32u;
    return f;
}
__device__ __forceinline__ void fast_divmod(int& q, int& r, int nsrc, FastDiv f) {
    q = (f.div != 1) ? (int)(__umulhi((unsigned int)nsrc, f.mul) >> f.shr) : nsrc;
    r = nsrc - q * f.div;
}
struct MegaDiv { FastDiv mn; FastDiv fast; FastDiv tail; };
